# V2 3D Tunnels and Trail-Riding Body: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task by task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Underground tunnels become real 3D half-pipes carved into the cut face, and the
burrowing worm's body always stays inside its tunnel. This ships as an emergency fix to
V2 · Underground.

**Architecture:**
- **Body.** It becomes a "chain on a rail": every body point sits on the head's recorded trail
  (a pursuit-smoothed path) at an exact chord distance behind the point ahead.
- **Tunnel.** It becomes a fine depth map (`hollow`) in the soil grid, written by the digs and
  castings. The renderer draws it as a pool of dense mesh tiles pushed back behind the face, all
  in one draw call. The face shader cuts a hole where the tunnel is, and both draws share one
  soil colour shader through a `HOLLOW` define.

**Tech stack:** TypeScript (strict), three r186, Vitest, Playwright, Vite. Spec:
`docs/superpowers/specs/2026-10-08-v2-3d-tunnels-design.md`.

### Task 1: `bodyTrail.ts`, the trail and the chain on a rail (pure)

**Files:**
- Create: `src/underground/bodyTrail.ts`
- Test: `tests/unit/bodyTrail.test.ts`

Concepts:
- **Trail.** A polyline of (u, v) points from oldest to newest, with cumulative arc length. Point
  `n-1` is the *live* pursuit point. It follows the head at most `TRAIL_LAG` behind, and it is
  committed (a new live point begun) once it is `TRAIL_STEP` past the point before it.
- **Trail place.** A float `k + f` means the point on segment k→k+1 at fraction f.
- **Chain on a rail.** Body point i (1..15) is the first point on the trail, walking back from
  point i−1's place, whose straight-line distance from point i−1 is `len[i]`. When backing,
  the walk goes forward from point i+1.

```ts
// src/underground/bodyTrail.ts
import { SPACING, WORM_SEGMENTS } from '../game/crawl';
import { hypot2 } from '../core/vec';

/**
 * The burrowing body rides its head's trail (V2 emergency fix, spec 2026-10-08 section 3): a chain
 * on a rail. The trail is the head's path drawn through a pursuit point that follows the head at
 * most TRAIL_LAG behind (a tractrix: sharp corners, a turn on the spot or a slide along a stone,
 * come out rounded to about TRAIL_LAG). Every trail point lies within TRAIL_LAG of where the head
 * then was, inside the TUNNEL_R disc it dug there, so the trail is always in open tunnel. Each body
 * point lies on the trail at exactly its link's length (a chord) from the point ahead: the links
 * never stretch or shrink (5ede96b put points at arc length and averaged them: the body shrank to
 * 58% where the trail folded). Backing, the tail leads back along the trail and the rest is placed
 * forward from the point behind. Pure; allocates nothing.
 */
const N = WORM_SEGMENTS;
/** A new trail point is begun once the live one is this far (u) past the point before it. */
export const TRAIL_STEP = 0.02;
/** The pursuit point follows the head at most this far (u) behind: under TUNNEL_R (0.14). */
export const TRAIL_LAG = 0.1;
/** Most trail points kept (about 160 u of path); when full the oldest half is dropped. */
export const TRAIL_CAP = 8192;
/** Seeded links ease to SPACING at this share per unit the head moves. */
const LEN_EASE = 2;

/** The body the trail poses: points head first, now and at the last step. */
export interface TrailBody {
  readonly u: Float64Array;
  readonly v: Float64Array;
  readonly prevU: Float64Array;
  readonly prevV: Float64Array;
}

export interface BodyTrail {
  /** Trail points, oldest first; point n - 1 is the live pursuit point. */
  readonly u: Float64Array;
  readonly v: Float64Array;
  /** Arc length from point 0. */
  readonly arc: Float64Array;
  n: number;
  /** Each body point's place on the trail (k + f: on segment k..k+1 at f). The head's only while backed. */
  readonly at: Float64Array;
  /** len[i]: the link from point i - 1 to point i (i >= 1). */
  readonly len: Float64Array;
  /** The head has backed along the trail since it last moved forward (its place is at[0]). */
  backed: boolean;
}

export function createTrail(): BodyTrail {
  return {
    u: new Float64Array(TRAIL_CAP), v: new Float64Array(TRAIL_CAP), arc: new Float64Array(TRAIL_CAP), n: 0,
    at: new Float64Array(N), len: new Float64Array(N).fill(SPACING), backed: false,
  };
}

function setArc(t: BodyTrail, k: number): void {
  t.arc[k] = k === 0 ? 0 : t.arc[k - 1] + hypot2(t.u[k] - t.u[k - 1], t.v[k] - t.v[k - 1]);
}

/** Appends (u, v) as the new last point. */
function push(t: BodyTrail, u: number, v: number): void {
  if (t.n === TRAIL_CAP) dropHalf(t);
  const k = t.n++;
  t.u[k] = u;
  t.v[k] = v;
  setArc(t, k);
}

/** Drops the oldest half (the body needs only its last 3.3 u; backing rarely more). */
function dropHalf(t: BodyTrail): void {
  const h = t.n >> 1;
  t.u.copyWithin(0, h, t.n);
  t.v.copyWithin(0, h, t.n);
  t.arc.copyWithin(0, h, t.n);
  t.n -= h;
  const a0 = t.arc[0];
  for (let k = 0; k < t.n; k++) t.arc[k] -= a0;
  for (let i = 0; i < N; i++) t.at[i] = Math.max(0, t.at[i] - h);
}

/**
 * A fresh trail through the body's points, tail first, with a live point at the head; the links
 * keep their lengths (eased to SPACING as the worm moves: nothing pops).
 */
export function seedTrail(t: BodyTrail, b: Pick<TrailBody, 'u' | 'v'>): void {
  t.n = 0;
  for (let i = N - 1; i >= 0; i--) {
    push(t, b.u[i], b.v[i]);
    t.at[i] = N - 1 - i;
  }
  push(t, b.u[0], b.v[0]);
  for (let i = 1; i < N; i++) t.len[i] = Math.max(1e-3, hypot2(b.u[i] - b.u[i - 1], b.v[i] - b.v[i - 1]));
  t.backed = false;
}

/** The point at place s, into `pt`. */
const pt = { u: 0, v: 0 };
function pointAt(t: BodyTrail, s: number): void {
  const k = Math.min(t.n - 1, Math.floor(s));
  const f = s - k;
  if (f <= 0 || k >= t.n - 1) {
    pt.u = t.u[k];
    pt.v = t.v[k];
    return;
  }
  pt.u = t.u[k] + (t.u[k + 1] - t.u[k]) * f;
  pt.v = t.v[k] + (t.v[k + 1] - t.v[k]) * f;
}

/**
 * After the head moved forward: if it had backed, the trail is cut at its place (the old branch
 * stays in the soil, not in the record); the pursuit point is drawn to within TRAIL_LAG of the
 * head and committed every TRAIL_STEP; the links ease toward SPACING.
 */
export function followHead(t: BodyTrail, b: TrailBody): void {
  if (t.backed) {
    pointAt(t, t.at[0]);
    t.n = Math.min(t.n, Math.floor(t.at[0]) + 1);
    push(t, pt.u, pt.v);
    t.backed = false;
  }
  const k = t.n - 1;
  const du = b.u[0] - t.u[k];
  const dv = b.v[0] - t.v[k];
  const d = hypot2(du, dv);
  if (d > TRAIL_LAG) {
    t.u[k] += (du * (d - TRAIL_LAG)) / d;
    t.v[k] += (dv * (d - TRAIL_LAG)) / d;
    setArc(t, k);
  }
  if (k > 0 && t.arc[k] - t.arc[k - 1] >= TRAIL_STEP) push(t, t.u[k], t.v[k]);
  const moved = hypot2(b.u[0] - b.prevU[0], b.v[0] - b.prevV[0]);
  const e = Math.min(1, moved * LEN_EASE);
  for (let i = 1; i < N; i++) t.len[i] += (SPACING - t.len[i]) * e;
}

/** Where the walks below put a point: (u, v) and its place s. */
const hit = { u: 0, v: 0, s: 0 };

/**
 * From place `s` (at the lead (qu, qv) itself, or nearer it than `len`), along the trail toward
 * `dir` (-1 back, +1 forward), the first place exactly `len` from the lead, into `hit`; the
 * trail's end when it gives out first.
 */
const lead = { u: 0, v: 0, s: 0, len: 0 };
function walk(t: BodyTrail, dir: number): void {
  pointAt(t, lead.s);
  let cu = pt.u;
  let cv = pt.v;
  let cs = lead.s;
  let j = dir < 0 ? Math.ceil(cs) - 1 : Math.floor(cs) + 1;
  const L2 = lead.len * lead.len;
  for (; j >= 0 && j < t.n; j += dir) {
    const bu = t.u[j];
    const bv = t.v[j];
    const mu = cu - lead.u;
    const mv = cv - lead.v;
    if ((bu - lead.u) * (bu - lead.u) + (bv - lead.v) * (bv - lead.v) >= L2) {
      const du = bu - cu;
      const dv = bv - cv;
      const a = du * du + dv * dv;
      if (a > 1e-18) {
        const bq = 2 * (du * mu + dv * mv);
        const c = mu * mu + mv * mv - L2;
        const r = Math.max(0, Math.min(1, (-bq + Math.sqrt(Math.max(0, bq * bq - 4 * a * c))) / (2 * a)));
        hit.u = cu + du * r;
        hit.v = cv + dv * r;
        hit.s = cs + (j - cs) * r;
        return;
      }
    }
    cu = bu;
    cv = bv;
    cs = j;
  }
  hit.u = cu;
  hit.v = cv;
  hit.s = cs;
}

/** Points 1..15 placed back along the trail from the head (the head itself is not on it). */
export function poseBehind(t: BodyTrail, b: TrailBody): void {
  for (let i = 1; i < N; i++) {
    lead.u = b.u[i - 1];
    lead.v = b.v[i - 1];
    // The head leads from the trail's live end; the others from their own place.
    lead.s = i === 1 ? t.n - 1 : t.at[i - 1];
    lead.len = t.len[i];
    if (i === 1) {
      // Walk from the live end, measured from the head (the live end is within TRAIL_LAG of it).
      walkFromEnd(t);
    } else walk(t, -1);
    b.u[i] = hit.u;
    b.v[i] = hit.v;
    t.at[i] = hit.s;
  }
}

/** Point 1: back from the live end, measured from the head (off the trail). */
function walkFromEnd(t: BodyTrail): void {
  const k = t.n - 1;
  const du = t.u[k] - lead.u;
  const dv = t.v[k] - lead.v;
  if (du * du + dv * dv >= lead.len * lead.len) {
    // Not expected (the live end is within TRAIL_LAG < SPACING of the head): toward the live end.
    const d = hypot2(du, dv);
    hit.u = lead.u + (du * lead.len) / d;
    hit.v = lead.v + (dv * lead.len) / d;
    hit.s = k;
    return;
  }
  // Treat the head as a point just past the live end: walk the trail back from the live end.
  const su = lead.u;
  const sv = lead.v;
  lead.s = k;
  walkFrom(t, su, sv);
}

/** walk(-1) from the trail point at place lead.s, measuring from (lead.u, lead.v) = (qu, qv). */
function walkFrom(t: BodyTrail, qu: number, qv: number): void {
  lead.u = qu;
  lead.v = qv;
  walk(t, -1);
}

/**
 * One step backing `step.step` (u): the head is pinned as the trail's end (no jump), the tail goes
 * back along the trail (never past its start: the tunnel's end), and the rest is placed forward
 * from the point behind, the head last.
 */
export function backAlong(t: BodyTrail, b: TrailBody, step: { step: number }): void {
  if (!t.backed) {
    push(t, b.u[0], b.v[0]);
    t.at[0] = t.n - 1;
    t.backed = true;
  }
  const T = N - 1;
  // The tail's place, moved back by `step` of arc.
  let s = t.at[T];
  let k = Math.min(t.n - 2, Math.floor(s));
  const seg = t.arc[k + 1] - t.arc[k];
  let want = t.arc[k] + seg * (s - k) - step.step;
  if (want <= 0) s = 0;
  else {
    while (k > 0 && t.arc[k] > want) k--;
    const l = t.arc[k + 1] - t.arc[k];
    s = k + (l > 1e-12 ? (want - t.arc[k]) / l : 0);
  }
  t.at[T] = s;
```

```ts
  out.c0 = d.c0; out.c1 = d.c1; out.r0 = d.r0; out.r1 = d.r1;

/** The hollow's depth (u) at (u, v), the nearest sample (tests and later interactions). */
export function hollowAt(g: SoilGrid, u: number, v: number): number {
  const c = Math.floor((u + SLAB_HALF) / HOLLOW_STEP);
  const r = Math.floor((v - g.v0) / HOLLOW_STEP);
  if (c < 0 || c >= HOLLOW_W || r < 0 || r >= HOLLOW_H) return 0;
  return (g.hollow[r * HOLLOW_W + c] * HOLLOW_MAX) / 255;

/**
 * Carves the hollow for a dig of radius r at `p`: each sample within r goes at least sqrt(r^2 - d^2)
 * deep (the back half of a round tunnel), but not in air or a blocker's cell. Allocates nothing.
 */
function carve(g: SoilGrid, p: UV, r: number): void {
  const u = p.u;
  const v = p.v;
  const c0 = Math.max(0, Math.floor((u - r + SLAB_HALF) / HOLLOW_STEP));
  const c1 = Math.min(HOLLOW_W - 1, Math.floor((u + r + SLAB_HALF) / HOLLOW_STEP));
  const r0 = Math.max(0, Math.floor((v - r - g.v0) / HOLLOW_STEP));
  const r1 = Math.min(HOLLOW_H - 1, Math.floor((v + r - g.v0) / HOLLOW_STEP));
  const scale = 255 / HOLLOW_MAX;
  for (let row = r0; row <= r1; row++) {
    const dv = g.v0 + (row + 0.5) * HOLLOW_STEP - v;
    const cellRow = (row >> 2) * GRID_W;
    for (let c = c0; c <= c1; c++) {
      const du = -SLAB_HALF + (c + 0.5) * HOLLOW_STEP - u;
      const d2 = du * du + dv * dv;
      if (d2 >= r * r) continue;
      const ci = cellRow + (c >> 2);
      if (g.layer[ci] === AIR || g.solid[ci] !== 0) continue;
      const depth = Math.min(255, Math.round(Math.sqrt(r * r - d2) * scale));
      const i = row * HOLLOW_W + c;
      if (depth > g.hollow[i]) {
        g.hollow[i] = depth;
        markHollow(g, c, row);
      }
    }
  }
}
```

```ts
 * Fills the hollow samples of cell i up to `level` (u behind the face) at most: a casting's mound.
 * `dist01` (0 at the casting's centre .. 1 at its edge) raises the mound toward its middle.
 */
export function fillHollowCell(g: SoilGrid, i: number, dist01: number): void {
  const cr = Math.floor(i / GRID_W);
  const cc = i - cr * GRID_W;
  const scale = 255 / HOLLOW_MAX;
  for (let a = 0; a < HOLLOW_RES; a++) {
    const row = cr * HOLLOW_RES + a;

        g.hollow[k] = level;
        markHollow(g, c, row);
      }
    }
  }
```

```ts
// src/underground/hollowMesh.ts
import * as THREE from 'three';
import { BOUNDS, B_TOP, B_F, B_H, B_A, B_E, B_B, SLAB_HALF, boundaryOf, bound, slabX, slabZ, type Slab } from './slab';
import { GRID_H, CELL, HOLLOW_H, HOLLOW_MAX, HOLLOW_STEP, HOLLOW_W, takeHollowDirty, type HollowRect, type SoilGrid } from './soilGrid';
import { CUT_FRAGMENT, CUT_VERTEX } from './cutFaceGlsl';

/**
 * The tunnels in 3D (spec 2026-10-08 section 4.3): the hollow (soilGrid.ts) drawn as fine tiles
 * pushed back behind the face by its depth. One mesh, one draw: a pool of TILE_U square tiles over
 * the soil grid, every slot built at load and claimed in turn as tunnels reach a tile (the draw
 * range covers the claimed slots). Where a tile has no hollow it lies BEHIND the face, hidden;
 * where the face cuts its hole (cutFaceGlsl.ts) the tile shows the tunnel's back half, lit by its
 * own normals, in the same soil colours as the face (one shader, HOLLOW defined). Only tiles a dig
 * changed are rewritten, each uploaded as its own range. Allocates nothing after load.
 */
export const TILE_U = 0.5;
/** Quads along a tile's side: 1/24 u apart, about 7 across a tunnel. */
export const TILE_QUADS = 12;
const TV = TILE_QUADS + 1;
export const VERTS_PER_TILE = TV * TV;
export const INDICES_PER_TILE = TILE_QUADS * TILE_QUADS * 6;
export const TILES_W = Math.ceil((2 * SLAB_HALF) / TILE_U);
export const TILES_H = Math.ceil((GRID_H * CELL) / TILE_U);
const SLOTS = TILES_W * TILES_H;
/** Where there is no hollow, a tile lies this far (u) behind the face, hidden by it. */
export const BEHIND = 0.004;
/** A tile is claimed for hollow within this (u) of its edges: the face's hole reaches a little past the dig. */
const CLAIM_MARGIN = 0.1;
const QUAD = TILE_U / TILE_QUADS;

export interface HollowMesh {
  readonly mesh: THREE.Mesh;
  /** A new slab: every slot freed (the hollow is all zero; the next upload claims nothing). */
  setSlab(s: Slab, g: SoilGrid): void;
  /** Claims and rewrites the tiles the hollow changed in since the last call. */
  upload(g: SoilGrid): void;
  /** Slot of each tile (row-major from the grid's bottom left), -1 unclaimed. */
  readonly slotOf: Int16Array;
  used(): number;
}

export function createHollowMesh(uniforms: Record<string, THREE.IUniform>): HollowMesh {
  const nv = SLOTS * VERTS_PER_TILE;
  const geo = new THREE.BufferGeometry();
  const pos = new THREE.BufferAttribute(new Float32Array(nv * 3), 3);
  const nrm = new THREE.BufferAttribute(new Float32Array(nv * 3), 3);
  const depth = new THREE.BufferAttribute(new Float32Array(nv), 1);
  const plane = new THREE.BufferAttribute(new Float32Array(nv * 2), 2);
  const ba = new THREE.BufferAttribute(new Float32Array(nv * 4), 4);
  const bb = new THREE.BufferAttribute(new Float32Array(nv * 2), 2);
  const attrs = [pos, nrm, depth, plane, ba, bb] as const;
  for (const a of attrs) a.setUsage(THREE.DynamicDrawUsage);
  const index = new Uint32Array(SLOTS * INDICES_PER_TILE);
  for (let slot = 0, o = 0; slot < SLOTS; slot++) {
    const base = slot * VERTS_PER_TILE;
    for (let y = 0; y < TILE_QUADS; y++) {
      for (let x = 0; x < TILE_QUADS; x++) {
        const a = base + y * TV + x;
        index[o++] = a; index[o++] = a + 1; index[o++] = a + TV;
        index[o++] = a + TV; index[o++] = a + 1; index[o++] = a + TV + 1;
      }
    }
  }
  geo.setAttribute('position', pos);
  geo.setAttribute('aSlabNormal', nrm);
  geo.setAttribute('aDepth', depth);
  geo.setAttribute('aPlane', plane);
  geo.setAttribute('aBoundsA', ba);
  geo.setAttribute('aBoundsB', bb);
  geo.setIndex(new THREE.BufferAttribute(index, 1));
  geo.setDrawRange(0, 0);
  geo.boundingSphere = new THREE.Sphere();

  // One fixed update range per slot and attribute (queued once until the renderer uploads).
  const ranges = attrs.map((a) => Array.from({ length: SLOTS }, (_, k) => ({ start: k * VERTS_PER_TILE * a.itemSize, count: VERTS_PER_TILE * a.itemSize })));
  const queued = attrs.map(() => new Uint8Array(SLOTS));
  attrs.forEach((a, j) => { a.onUpload(() => queued[j].fill(0)); });
  function queue(j: number, slot: number): void {
    if (queued[j][slot]) return;
    queued[j][slot] = 1;
    attrs[j].updateRanges.push(ranges[j][slot]);
    attrs[j].needsUpdate = true;
  }

  const material = new THREE.ShaderMaterial({
    uniforms,
    defines: { HOLLOW: '' },
    vertexShader: CUT_VERTEX,
    fragmentShader: CUT_FRAGMENT,
    side: THREE.DoubleSide,
  });
  const mesh = new THREE.Mesh(geo, material);
  mesh.name = 'cutHollow';

  const slotOf = new Int16Array(SLOTS).fill(-1);
  const tileOfSlot = new Int16Array(SLOTS);
  let used = 0;
  let slab: Slab | null = null;
  const dirty: HollowRect = { c0: 0, c1: -1, r0: 0, r1: -1 };
  const at = { u: 0, v: 0 };

  /** The hollow's depth (u) at (u, v), bilinear between samples; 0 outside the map. */
  function depthAt(g: SoilGrid, u: number, v: number): number {
    const fx = (u + SLAB_HALF) / HOLLOW_STEP - 0.5;
    const fy = (v - g.v0) / HOLLOW_STEP - 0.5;
    const x0 = Math.floor(fx);
    const y0 = Math.floor(fy);
    const tx = fx - x0;
    const ty = fy - y0;
    const s00 = sample(g, x0, y0);
    const s10 = sample(g, x0 + 1, y0);
    const s01 = sample(g, x0, y0 + 1);
    const s11 = sample(g, x0 + 1, y0 + 1);
    return (((s00 * (1 - tx) + s10 * tx) * (1 - ty) + (s01 * (1 - tx) + s11 * tx) * ty) * HOLLOW_MAX) / 255;
  }
  function sample(g: SoilGrid, c: number, r: number): number {
    return c < 0 || c >= HOLLOW_W || r < 0 || r >= HOLLOW_H ? 0 : g.hollow[r * HOLLOW_W + c];
  }

  /** Writes slot `slot`'s depths, positions and normals from the hollow (tile `tile`). */
  function writeDepths(g: SoilGrid, slot: number, tile: number): void {
    const s = slab!;
    const tu = (tile % TILES_W) * TILE_U - SLAB_HALF;
    const tv = Math.floor(tile / TILES_W) * TILE_U + g.v0;
    const P = pos.array as Float32Array;
    const Nn = nrm.array as Float32Array;
    const D = depth.array as Float32Array;
    const h = HOLLOW_STEP;
    for (let y = 0; y < TV; y++) {
      for (let x = 0; x < TV; x++) {
        const k = slot * VERTS_PER_TILE + y * TV + x;
        const u = tu + x * QUAD;
        const v = tv + y * QUAD;
        const d = depthAt(g, u, v);
        const w = Math.max(BEHIND, d);
        P[k * 3] = slabX(s, u) - s.nx * w;
        P[k * 3 + 1] = v;
        P[k * 3 + 2] = slabZ(s, u) - s.nz * w;
        D[k] = d;
        // The surface is w = -depth(u, v): its normal (slab frame: u, v, toward the camera) is
        // (dDepth/du, dDepth/dv, 1), normalised.
        const gu = (depthAt(g, u + h, v) - depthAt(g, u - h, v)) / (2 * h);
        const gv = (depthAt(g, u, v + h) - depthAt(g, u, v - h)) / (2 * h);
        const l = Math.sqrt(gu * gu + gv * gv + 1);
        Nn[k * 3] = gu / l;
        Nn[k * 3 + 1] = gv / l;
        Nn[k * 3 + 2] = 1 / l;
      }
    }
    queue(0, slot);
    queue(1, slot);
    queue(2, slot);
  }

  /** Claims tile `tile` into the next slot: its plane points and layer boundaries, then its depths. */
  function claim(g: SoilGrid, tile: number): void {
    const s = slab!;
    const slot = used++;
    slotOf[tile] = slot;
    tileOfSlot[slot] = tile;
    const tu = (tile % TILES_W) * TILE_U - SLAB_HALF;
    const tv = Math.floor(tile / TILES_W) * TILE_U + g.v0;
    const Q = plane.array as Float32Array;
    const A = ba.array as Float32Array;
    const B = bb.array as Float32Array;
    for (let x = 0; x < TV; x++) {
      at.u = tu + x * QUAD;
      boundaryOf(s, B_TOP, at); const top = bound.v;
      boundaryOf(s, B_F, at); const f = bound.v;
      boundaryOf(s, B_H, at); const hh = bound.v;
      boundaryOf(s, B_A, at); const a = bound.v;
      boundaryOf(s, B_E, at); const e = bound.v;
      boundaryOf(s, B_B, at); const b = bound.v;
      for (let y = 0; y < TV; y++) {
        const k = slot * VERTS_PER_TILE + y * TV + x;
        Q[k * 2] = at.u;
        Q[k * 2 + 1] = tv + y * QUAD;
        A[k * 4] = top; A[k * 4 + 1] = f; A[k * 4 + 2] = hh; A[k * 4 + 3] = a;
        B[k * 2] = e; B[k * 2 + 1] = b;
      }
    }
    queue(3, slot);
    queue(4, slot);
    queue(5, slot);
    writeDepths(g, slot, tile);
    geo.setDrawRange(0, used * INDICES_PER_TILE);
  }

  /** Any hollow sample in tile `tile`, its edges widened by CLAIM_MARGIN. */
  function needs(g: SoilGrid, tile: number): boolean {
    const m = Math.round(CLAIM_MARGIN / HOLLOW_STEP);
    const per = Math.round(TILE_U / HOLLOW_STEP);
    const c0 = Math.max(0, (tile % TILES_W) * per - m);
    const r0 = Math.max(0, Math.floor(tile / TILES_W) * per - m);
    const c1 = Math.min(HOLLOW_W - 1, c0 + per + 2 * m);
    const r1 = Math.min(HOLLOW_H - 1, r0 + per + 2 * m);
    for (let r = r0; r <= r1; r++) {
      const o = r * HOLLOW_W;
      for (let c = c0; c <= c1; c++) if (g.hollow[o + c]) return true;
    }
    return false;
  }

  return {
    mesh,
    slotOf,
    used: () => used,
    setSlab(s, g) {
      slab = s;
      slotOf.fill(-1);
      used = 0;
      geo.setDrawRange(0, 0);
      // The same bounds as the face (cutFace.ts): the whole playable slab, generously.
      geo.boundingSphere!.center.set(slabX(s, 0), g.v0 + (GRID_H * CELL) / 2, slabZ(s, 0));
      geo.boundingSphere!.radius = Math.hypot(SLAB_HALF, (GRID_H * CELL) / 2) + 1;
    },
    upload(g) {
      takeHollowDirty(g, dirty);
      if (dirty.c0 > dirty.c1 || !slab) return;
      const per = TILE_U / HOLLOW_STEP;
      const m = CLAIM_MARGIN / HOLLOW_STEP;
      const t0 = Math.max(0, Math.floor((dirty.c0 - m) / per));
      const t1 = Math.min(TILES_W - 1, Math.floor((dirty.c1 + m) / per));
      const s0 = Math.max(0, Math.floor((dirty.r0 - m) / per));
      const s1 = Math.min(TILES_H - 1, Math.floor((dirty.r1 + m) / per));
      for (let ty = s0; ty <= s1; ty++) {
        for (let tx = t0; tx <= t1; tx++) {
          const tile = ty * TILES_W + tx;
          const slot = slotOf[tile];
          if (slot >= 0) writeDepths(g, slot, tile);
          else if (needs(g, tile)) claim(g, tile);
        }
      }
    },
  };
}
```
